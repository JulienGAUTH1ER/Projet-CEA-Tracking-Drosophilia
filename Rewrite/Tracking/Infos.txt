    '''def contrast_larva(self, frame):
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        diff = cv2.absdiff(gray, self.get_background())
        cv2.namedWindow("Diff brute", cv2.WINDOW_NORMAL)
        cv2.imshow("Diff brute", diff)

        diff = cv2.normalize(diff, None, 0, 255, cv2.NORM_MINMAX)
        cv2.namedWindow("Diff normalisée", cv2.WINDOW_NORMAL)
        cv2.imshow("Diff normalisée", diff)

        diff = cv2.convertScaleAbs(diff, alpha=self.alpha_contrast)
        cv2.namedWindow("Diff contrastée", cv2.WINDOW_NORMAL)
        cv2.imshow("Diff contrastée", diff)

        _, thresh = cv2.threshold(diff, self.threshold_value, 255, cv2.THRESH_BINARY)
        cv2.namedWindow("Seuil", cv2.WINDOW_NORMAL)
        cv2.imshow("Seuil", thresh)

        cv2.waitKey(1)

        return thresh'''
